javascript const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); app.use(express.static(__dirname + '/public')); let acikOdalar = ['Genel', '#Sohbet', '#Oyun', '#Yardım']; let aktifKullanicilar = {}; let banliListesi = []; io.on('connection', (socket) => { socket.emit('oda listesi guncelle', acikOdalar); socket.on('profil guncelle', (data) => { if (banliListesi.includes(data.nick)) { socket.emit('sistem cezasi', 'Yasaklandınız! Bu chat odasına girişiniz engellendi.'); socket.disconnect(); return; } let mevcutYetki = data.nick === 'Kral' ? 'Kurucu' : (aktifKullanicilar[socket.id]?.yetki || 'Kullanıcı'); aktifKullanicilar[socket.id] = { nick: data.nick, yetki: mevcutYetki }; io.emit('kullanici listesi guncelle', Object.values(aktifKullanicilar)); }); socket.on('odaya katil', (odaAdi) => { Array.from(socket.rooms).forEach(room => { if (room !== socket.id) socket.leave(room); }); socket.join(odaAdi); }); socket.on('yeni mesaj', (data) => { const kullaniciVerisi = aktifKullanicilar[socket.id] || { yetki: 'Kullanıcı' }; io.to(data.oda).emit('mesaj al', { kullanici: data.kullanici, metin: data.metin, yetki: kullaniciVerisi.yetki }); }); socket.on('yeni oda olustur', (yeniOdaAdi) => { const user = aktifKullanicilar[socket.id]; if (user && user.yetki === 'Kurucu' && !acikOdalar.includes(yeniOdaAdi)) { acikOdalar.push(yeniOdaAdi); io.emit('oda listesi guncelle', acikOdalar); } }); socket.on('yetki degistir', (data) => { const talepEden = aktifKullanicilar[socket.id]; if (!talepEden || talepEden.yetki !== 'Kurucu') return; const hedefSocketId = Object.keys(aktifKullanicilar).find(id => aktifKullanicilar[id].nick === data.hedefNick); if (hedefSocketId && aktifKullanicilar[hedefSocketId].yetki !== 'Kurucu') { aktifKullanicilar[hedefSocketId].yetki = data.yeniYetki; io.emit('kullanici listesi guncelle', Object.values(aktifKullanicilar)); io.emit('sistem mesajı yayini', `⚙️ ${data.hedefNick} yetkisi [${data.yeniYetki}] yapıldı.`); } }); socket.on('kullanici banla', (hedefNick) => { const talepEden = aktifKullanicilar[socket.id]; if (!talepEden || talepEden.yetki !== 'Kurucu' || hedefNick === 'Kral') return; if (!banliListesi.includes(hedefNick)) banliListesi.push(hedefNick); const hedefSocketId = Object.keys(aktifKullanicilar).find(id => aktifKullanicilar[id].nick === hedefNick); if (hedefSocketId) { io.to(hedefSocketId).emit('sistem cezasi', 'Kurucu tarafından banlandınız!'); io.sockets.sockets.get(hedefSocketId)?.disconnect(); } io.emit('sistem mesajı yayini', `🚨 ${hedefNick} chattan kalıcı olarak banlandı!`); }); socket.on('disconnect', () => { if (aktifKullanicilar[socket.id]) { delete aktifKullanicilar[socket.id]; io.emit('kullanici listesi guncelle', Object.values(aktifKullanicilar)); } }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => console.log(`Kral Chat http://localhost:${PORT} üzerinde aktif!`)); Kodu dikkatli kullanın. 2. public Klasörünü ve index.html Dosyasını Düzenleyin: • Ana klasörün içine public isimli bir klasör oluşturun. • public klasörünün içine girip index.html dosyasını oraya taşıyın/oluşturun. • Ekranınızdaki o tek satırlık karmaşık kodların tamamını silin ve yerine sadece temiz olan şu HTML kodunu yapıştırın: html Kral Chat - Yönetim Paneli
