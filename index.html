const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); // İstemci dosyalarının sunulacağı klasör app.use(express.static(__dirname + '/public')); // Socket.io bağlantı yönetimi io.on('connection', (socket) => { console.log('Bir kullanıcı bağlandı:', socket.id); // Kullanıcıdan gelen mesajı al ve herkese yayınla socket.on('chat message', (data) => { io.emit('chat message', { user: data.user, text: data.text, time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }); }); // Kullanıcı ayrıldığında socket.on('disconnect', () => { console.log('Bir kullanıcı ayrıldı:', socket.id); }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Sunucu http://localhost:${PORT} adresinde çalışıyor...`); });
