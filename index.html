const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); app.use(express.static(__dirname + '/public')); let acikOdalar = ['Genel', '#Sohbet', '#Oyun', '#Yardım']; let aktifKullanicilar = {}; // socket.id -> { nick, yetki } let banliListesi = []; // Banlanan IP veya kullanıcı takibi için io.on('connection', (socket) => { console.log('Yeni bağlantı:', socket.id); // Kullanıcı ilk bağlandığında oda listesini gönder socket.emit('oda listesi guncelle', acikOdalar); // Kullanıcı rumuzunu güncellediğinde veya ilk girdiğinde socket.on('profil guncelle', (data) => { // Ban kontrolü if (banliListesi.includes(data.nick)) { socket.emit('sistem cezasi', 'Yasaklandınız! Bu chat odasına girişiniz engellendi.'); socket.disconnect(); return; } // Eğer önceden kayıtlı değilse varsayılan yetki 'Kullanıcı' (Kral ise 'Kurucu') let mevcutYetki = data.nick === 'Kral' ? 'Kurucu' : (aktifKullanicilar[socket.id]?.yetki || 'Kullanıcı'); aktifKullanicilar[socket.id] = { nick: data.nick, yetki: mevcutYetki }; // Güncel listeyi herkese duyur io.emit('kullanici listesi guncelle', Object.values(aktifKullanicilar)); }); // Odaya katılma isteği socket.on('odaya katil', (odaAdi) => { Array.from(socket.rooms).forEach(room => { if (room !== socket.id) socket.leave(room); }); socket.join(odaAdi); }); // Mesajlaşma socket.on('yeni mesaj', (data) => { const kullaniciVerisi = aktifKullanicilar[socket.id] || { yetki: 'Kullanıcı' }; io.to(data.oda).emit('mesaj al', { kullanici: data.kullanici, metin: data.metin, yetki: kullaniciVerisi.yetki }); }); // Yeni oda oluşturma (Sadece Kurucu) socket.on('yeni oda olustur', (yeniOdaAdi) => { const user = aktifKullanicilar[socket.id]; if (user && user.yetki === 'Kurucu' && !acikOdalar.includes(yeniOdaAdi)) { acikOdalar.push(yeniOdaAdi); io.emit('oda listesi guncelle', acikOdalar); } }); // YÖNETİM: Yetki Verme / Alma socket.on('yetki degistir', (data) => { const talepEden = aktifKullanicilar[socket.id]; if (!talepEden || talepEden.yetki !== 'Kurucu') return; // Sadece Kurucu yapabilir // Hedef kullanıcıyı bul const hedefSocketId = Object.keys(aktifKullanicilar).find(id => aktifKullanicilar[id].nick === data.hedefNick); if (hedefSocketId && aktifKullanicilar[hedefSocketId].yetki !== 'Kurucu') { aktifKullanicilar[hedefSocketId].yetki = data.yeniYetki; io.emit('kullanici listesi guncelle', Object.values(aktifKullanicilar)); io.emit('sistem mesajı yayini', `⚙️ ${data.hedefNick} isimli kullanıcının yetkisi [${data.yeniYetki}] olarak güncellendi.`); } }); // YÖNETİM: Ban Atma socket.on('kullanici banla', (hedefNick) => { const talepEden = aktifKullanicilar[socket.id]; if (!talepEden || talepEden.yetki !== 'Kurucu') return; if (hedefNick === 'Kral') return; // Kurucu kendini banlayamaz if (!banliListesi.includes(hedefNick)) { banliListesi.push(hedefNick); } const hedefSocketId = Object.keys(aktifKullanicilar).find(id => aktifKullanicilar[id].nick === hedefNick); if (hedefSocketId) { io.to(hedefSocketId).emit('sistem cezasi', 'Kurucu tarafından odadan kalıcı olarak uçuruldunuz!'); io.sockets.sockets.get(hedefSocketId)?.disconnect(); } io.emit('sistem mesajı yayini', `🚨 ${hedefNick} chattan kalıcı olarak banlandı!`); }); // Bağlantı koptuğunda listeden sil socket.on('disconnect', () => { if (aktifKullanicilar[socket.id]) { delete aktifKullanicilar[socket.id]; io.emit('kullanici listesi guncelle', Object.values(aktifKullanicilar)); } }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => console.log(`Kral Chat http://localhost:${PORT} üzerinde aktif!`)); Kodu dikkatli kullanın. 2.
